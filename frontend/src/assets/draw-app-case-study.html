3. Mathematical Foundation: Screen Space ↔ Canvas Viewport Transformation

When panning and zooming across an infinite whiteboard, mouse coordinates $(X_{screen}, Y_{screen})$ must be mapped to world space coordinates $(X_{world}, Y_{world})$.

// Coordinate System Transformation Matrix
export function screenToCanvasCoords(
  screenX: number, 
  screenY: number, 
  panOffset: { x: number; y: number }, 
  zoomScale: number
): { x: number; y: number } {
  return {
    x: (screenX - panOffset.x) / zoomScale,
    y: (screenY - panOffset.y) / zoomScale,
  };
}
